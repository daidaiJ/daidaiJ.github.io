{{/* 按 seed 把主源打散到 met / unsplash / picsum；同图走 wsrv 备份，跨源做故障降级。 */}} {{- $permalink := .permalink | default "" -}} {{- $seed := .seed | default "" -}} {{- $arts := site.Data.cover_art.paintings | default slice -}} {{- $unsplashPool := site.Data.cover_art.unsplash | default slice -}} {{- if not $seed -}} {{- $u := urls.Parse $permalink -}} {{- $parts := split $u.Path "/" -}} {{- if and (ge (len $parts) 3) (eq (index $parts 1) "seed") -}} {{- $seed = index $parts 2 -}} {{- end -}} {{- end -}} {{- if not $seed -}} {{- $seed = substr (md5 $permalink) 4 8 -}} {{- end -}} {{- $digits := replaceRE "[^0-9]" "" (md5 $seed) -}} {{- if lt (len $digits) 1 -}}{{- $digits = "1" -}}{{- end -}} {{- $srcRaw := replaceRE "^0+" "" (substr (printf "%s1" $digits) 0 4) -}} {{- if eq $srcRaw "" -}}{{- $srcRaw = "1" -}}{{- end -}} {{- $artRaw := replaceRE "^0+" "" (substr (printf "%s9" $digits) 4 4) -}} {{- if eq $artRaw "" -}}{{- $artRaw = "1" -}}{{- end -}} {{- $srcNum := int $srcRaw -}} {{- $artNum := int $artRaw -}} {{- $names := slice "met" "unsplash" "picsum" -}} {{- $primary := index $names (mod $srcNum 3) -}} {{- $met := $permalink -}} {{- if gt (len $arts) 0 -}} {{- $met = index $arts (mod $artNum (len $arts)) -}} {{- end -}} {{- $unsplash := "" -}} {{- if gt (len $unsplashPool) 0 -}} {{- $unsplash = index $unsplashPool (mod $artNum (len $unsplashPool)) -}} {{- end -}} {{- $picsum := printf "https://picsum.photos/seed/%s/800/600.webp" $seed -}} {{- $primaryUrl := $picsum -}} {{- if eq $primary "met" -}} {{- $primaryUrl = $met -}} {{- else if and (eq $primary "unsplash") $unsplash -}} {{- $primaryUrl = $unsplash -}} {{- end -}} {{- $origin := replace (replace $primaryUrl "https://" "") "http://" "" -}} {{- $origin = replace (replace $origin "?" "%3F") "&" "%26" -}} {{- $wsrv := printf "https://wsrv.nl/?url=%s&w=800&h=600&fit=cover&output=webp" $origin -}} {{- $weserv := printf "https://images.weserv.nl/?url=%s&w=800&h=600&fit=cover&output=webp" $origin -}} {{- return (dict "primary" $primary "primaryUrl" $primaryUrl "met" $met "unsplash" $unsplash "picsum" $picsum "wsrv" $wsrv "weserv" $weserv "seed" $seed) -}}
